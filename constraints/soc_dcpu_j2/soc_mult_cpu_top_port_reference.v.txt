// Clock/reset excerpt transcribed from photo5.jpg through photo8.jpg, Michael Tan, 20260803
// The original SoC path below depends on DDR MIG signals (ddr_ui_clk/ddr_ui_rst).
// It is reference material only and must not be copied into the DDR-free NoC top.
`ifdef SOC_SIM
    assign pad_cpu_rst_b = l_pad_rst_b;
    assign dirty_enable = 1'b1;
    assign wrsnp_enable = 1'b1;
    assign pref_enable = 1'b1;
    assign cache_enable = 1'b1;
`else
    wire l_pad_clk;
    wire ddr_ui_clk;
    wire ddr_ui_rst;
    wire c0_init_calib_complete;
    // assign o_ddr_init_led = c0_init_calib_complete;
    assign o_ddr_init_led = c0_init_calib_complete | crid_alarm0 | crid_alarm1 | cdid_alarm0 | cdid_alarm1;

    wire sys_clk;
    wire pll_rstn;
    wire vio_top_rst;

    IBUFDS #(
        .DIFF_TERM("FALSE"),
        .IBUF_LOW_PWR("FALSE")
    )
    u_ibufg_sys_clk (
        .I(l_pad_clk_p),
        .IB(l_pad_clk_n),
        .O(sys_clk)
    );

    xlnx_clkgen inst_soc_clkgen (
        .clk_out1(l_pad_clk), // output clk_out1
        .reset(ddr_ui_rst),   // input reset
        .locked(pll_rstn),    // output locked
        .clk_in1(ddr_ui_clk)  // input clk_in1
    );

    assign pad_cpu_rst_b = pll_rstn & vio_top_rst;

    VIO_Top u_vio_ctrl (
        .clk(l_pad_clk),
        .probe_in0(vio_top_rst),
        .probe_out0(vio_top_rst)
    );

    always @(posedge l_pad_clk)
    begin
        pref_enable <= 1'b1;
        cache_enable <= 1'b1;
    end
`endif
